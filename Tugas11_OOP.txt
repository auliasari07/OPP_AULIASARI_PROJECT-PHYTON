# Inisialisasi variabel untuk menyimpan total penjumlahan
total_jumlah = 0

# Perulangan dari angka 1 sampai 100
# range(1, 101) berarti perulangan dimulai dari 1 hingga angka sebelum 101, yaitu 100
for angka in range(1, 101):
    total_jumlah += angka

# Menampilkan hasil akhir
print("Halo Rina! Jumlah seluruh angka dari 1 sampai 100 adalah:", total_jumlah)
